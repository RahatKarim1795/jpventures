'use client';

import { ProjectWithImages, FileSystemImage, ProjectImage } from '@/lib/types';
import Container from '@/components/ui/Container';
import Image from 'next/image';
import { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'motion/react';
import { X, ChevronLeft, ChevronRight, Maximize2 } from 'lucide-react';

interface ProjectGalleryProps {
  project: ProjectWithImages;
  // New prop for file system images
  fileSystemImages?: FileSystemImage[];
}

function isFileSystemImage(img: ProjectImage | FileSystemImage): img is FileSystemImage {
  return (img as FileSystemImage).src !== undefined;
}

export default function ProjectGallery({ project, fileSystemImages }: ProjectGalleryProps) {
  const [selectedImage, setSelectedImage] = useState<number | null>(null);
  
  // Use fileSystemImages if available, otherwise fallback to project.images
  const images = fileSystemImages || (project.images as (ProjectImage | FileSystemImage)[]);
  const hasImages = images.length > 0;

  // Keyboard navigation with useEffect - must be called before any returns
  useEffect(() => {
    if (selectedImage === null || !hasImages) return;
    
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'ArrowRight') {
        setSelectedImage((selectedImage + 1) % images.length);
      } else if (e.key === 'ArrowLeft') {
        setSelectedImage((selectedImage - 1 + images.length) % images.length);
      } else if (e.key === 'Escape') {
        setSelectedImage(null);
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [selectedImage, images.length, hasImages]);

  if (!hasImages) return null;

  const getImageUrl = (img: ProjectImage | FileSystemImage) => {
    if (isFileSystemImage(img)) return img.src;
    // Fallback for DB images
    return img.filename.startsWith('/') 
      ? img.filename 
      : `/uploads/projects/${project.id}/${img.filename}`;
  };

  const getImageAlt = (img: ProjectImage | FileSystemImage, index: number) => {
    if (isFileSystemImage(img)) return img.alt;
    return img.altText || `Gallery image ${index + 1}`;
  };

  const getImageCaption = (img: ProjectImage | FileSystemImage) => {
    if (isFileSystemImage(img)) return img.caption;
    return img.caption;
  };

  const handleNext = (e: React.MouseEvent) => {
    e.stopPropagation();
    if (selectedImage !== null) {
      setSelectedImage((selectedImage + 1) % images.length);
    }
  };

  const handlePrev = (e: React.MouseEvent) => {
    e.stopPropagation();
    if (selectedImage !== null) {
      setSelectedImage((selectedImage - 1 + images.length) % images.length);
    }
  };

  return (
    <section className="relative py-20 bg-brand-midnight-blue">
      <Container>
        <div className="mb-12">
          <motion.h2 
            initial={{ opacity: 0, y: 20 }}
            whileInView={{ opacity: 1, y: 0 }}
            viewport={{ once: true }}
            className="text-3xl md:text-4xl font-bold text-brand-concrete mb-4"
          >
            Gallery
          </motion.h2>
          <p className="text-brand-concrete max-w-2xl">
            Visual journey through {project.title}
          </p>
        </div>

        {/* Grid Layout */}
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6 md:gap-8">
          {images.map((img, index) => {
            const caption = getImageCaption(img);
            
            return (
              <motion.div 
                key={isFileSystemImage(img) ? img.src : (img as ProjectImage).id}
                initial={{ opacity: 0 }}
                whileInView={{ opacity: 1 }}
                viewport={{ once: true }}
                transition={{ duration: 0.5 }}
                className="relative aspect-[4/3] overflow-hidden rounded-2xl group cursor-pointer shadow-lg hover:shadow-2xl transition-shadow duration-300"
                onClick={() => setSelectedImage(index)}
              >
                <Image
                  src={getImageUrl(img)}
                  alt={getImageAlt(img, index)}
                  fill
                  className="object-cover transition-transform duration-700 group-hover:scale-110"
                  sizes="(max-width: 768px) 100vw, (max-width: 1024px) 50vw, 33vw"
                />
                
                {/* Hover Overlay */}
                <div className="absolute inset-0 bg-black/0 group-hover:bg-black/30 transition-colors duration-300" />
                
                {/* Hover Icon */}
                <div className="absolute inset-0 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity duration-300">
                  <div className="bg-white/20 backdrop-blur-md p-4 rounded-full text-white shadow-lg transform group-hover:scale-110 transition-transform duration-300">
                    <Maximize2 size={32} />
                  </div>
                </div>
                
                {/* Caption if available */}
                {caption && (
                  <div className="absolute bottom-0 left-0 w-full p-4 bg-gradient-to-t from-black/80 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-300">
                    <p className="text-white text-sm md:text-base font-medium">{caption}</p>
                  </div>
                )}
              </motion.div>
            );
          })}
        </div>
      </Container>

      {/* Lightbox Modal */}
      <AnimatePresence>
        {selectedImage !== null && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="fixed inset-0 z-50 bg-black/95 flex items-center justify-center p-4"
            onClick={() => setSelectedImage(null)}
          >
            {/* Close Button */}
            <button 
              className="absolute top-6 right-6 text-white/70 hover:text-white p-2 z-10 transition-colors rounded-full hover:bg-white/10"
              onClick={() => setSelectedImage(null)}
              aria-label="Close gallery"
            >
              <X size={32} />
            </button>

            {/* Previous Button */}
            <button
              className="absolute left-4 md:left-8 top-1/2 -translate-y-1/2 text-white/70 hover:text-white p-3 transition-colors rounded-full hover:bg-white/10 z-10"
              onClick={handlePrev}
              aria-label="Previous image"
            >
              <ChevronLeft size={40} />
            </button>

            {/* Image Container */}
            <motion.div 
              className="relative w-full max-w-7xl max-h-[90vh] aspect-video"
              initial={{ scale: 0.9 }}
              animate={{ scale: 1 }}
              exit={{ scale: 0.9 }}
              transition={{ duration: 0.3 }}
              onClick={e => e.stopPropagation()}
            >
              <Image
                src={getImageUrl(images[selectedImage])}
                alt={getImageAlt(images[selectedImage], selectedImage)}
                fill
                className="object-contain"
                quality={100}
              />
              
              {/* Image Counter and Caption */}
              <div className="absolute bottom-4 left-0 right-0 text-center">
                <div className="inline-block bg-black/70 backdrop-blur-sm px-6 py-3 rounded-full">
                  <p className="text-white/90 text-sm font-medium mb-1">
                    {selectedImage + 1} / {images.length}
                  </p>
                  {getImageCaption(images[selectedImage]) && (
                    <p className="text-white font-medium text-base">
                      {getImageCaption(images[selectedImage])}
                    </p>
                  )}
                </div>
              </div>
            </motion.div>

            {/* Next Button */}
            <button
              className="absolute right-4 md:right-8 top-1/2 -translate-y-1/2 text-white/70 hover:text-white p-3 transition-colors rounded-full hover:bg-white/10 z-10"
              onClick={handleNext}
              aria-label="Next image"
            >
              <ChevronRight size={40} />
            </button>

            {/* Mobile Navigation Hint */}
            <div className="absolute bottom-20 left-1/2 -translate-x-1/2 text-white/50 text-sm md:hidden">
              Swipe or use arrows to navigate
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </section>
  );
}
