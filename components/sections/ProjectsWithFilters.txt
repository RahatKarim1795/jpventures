'use client';

import { ProjectWithFeaturedImage, ProjectStatus, ProjectType } from '@/lib/types';
import ProjectCard from '@/components/ui/ProjectCard';
import Container from '@/components/ui/Container';
import { useState, useMemo, useRef, useEffect } from 'react';
import { motion } from 'motion/react';

interface ProjectsWithFiltersProps {
  projects: ProjectWithFeaturedImage[];
}

interface FilterDropdownProps {
  label: string;
  value: string;
  options: string[];
  onChange: (value: string) => void;
}

const FilterDropdown = ({ label, value, options, onChange }: FilterDropdownProps) => {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const handleClickOutside = (event: MouseEvent) => {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target as Node)) {
        setIsOpen(false);
      }
    };

    document.addEventListener('mousedown', handleClickOutside);
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
    };
  }, []);

  return (
    <div className="relative" ref={dropdownRef}>
      <button
        onClick={() => setIsOpen(!isOpen)}
        className="flex items-center justify-between w-full md:w-48 px-4 py-3 bg-brand-midnight-blue border border-brand-dutch-blue rounded-lg text-white hover:border-brand-teal-sky transition-colors"
      >
        <span className="block truncate mr-2">
          {value === 'all' ? label : value}
        </span>
        <svg
          className={`w-4 h-4 transition-transform duration-200 ${isOpen ? 'rotate-180' : ''}`}
          fill="none"
          stroke="currentColor"
          viewBox="0 0 24 24"
        >
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
        </svg>
      </button>

      {isOpen && (
        <div className="absolute z-50 w-full mt-1 bg-brand-midnight-blue border border-brand-dutch-blue rounded-lg shadow-xl max-h-60 overflow-y-auto">
          <button
            onClick={() => {
              onChange('all');
              setIsOpen(false);
            }}
            className={`w-full text-left px-4 py-2 text-sm hover:bg-brand-dutch-blue transition-colors ${
              value === 'all' ? 'text-brand-teal-sky font-semibold' : 'text-gray-300'
            }`}
          >
            All {label}s
          </button>
          {options.map((option) => (
            <button
              key={option}
              onClick={() => {
                onChange(option);
                setIsOpen(false);
              }}
              className={`w-full text-left px-4 py-2 text-sm hover:bg-brand-dutch-blue transition-colors ${
                value === option ? 'text-brand-teal-sky font-semibold' : 'text-gray-300'
              }`}
            >
              {option}
            </button>
          ))}
        </div>
      )}
    </div>
  );
};

export default function ProjectsWithFilters({ projects }: ProjectsWithFiltersProps) {
  const [statusFilter, setStatusFilter] = useState<'all' | ProjectStatus>('all');
  const [typeFilter, setTypeFilter] = useState<'all' | ProjectType>('all');
  const [locationFilter, setLocationFilter] = useState<string>('all');

  // Extract unique locations
  const locations = useMemo(() => {
    const locs = new Set<string>();
    projects.forEach(p => {
      // Simple heuristic: extract part after first comma, or use a default
      // If address structure is varied, this might need adjustment
      // For now, assume comma separated or just take the whole string if short
      // Or better: just treat the whole address as unique key if short enough, 
      // or try to find "Bashundhara R/A" etc.

      if (p.address.toLowerCase().includes('bashundhara')) {
        locs.add('Bashundhara R/A');
      } else {
        // Fallback to using the full address or a simplified version
        locs.add(p.address);
      }
    });
    return Array.from(locs).sort();
  }, [projects]);

  // Filter projects
  const filteredProjects = useMemo(() => {
    return projects.filter(project => {
      const matchesStatus = statusFilter === 'all' || project.status === statusFilter;
      const matchesType = typeFilter === 'all' || project.type === typeFilter;
      
      let matchesLocation = locationFilter === 'all';
      if (!matchesLocation) {
        if (locationFilter === 'Bashundhara R/A') {
           matchesLocation = project.address.toLowerCase().includes('bashundhara');
        } else {
           matchesLocation = project.address === locationFilter;
        }
      }

      return matchesStatus && matchesType && matchesLocation;
    });
  }, [projects, statusFilter, typeFilter, locationFilter]);

  // Reset filters
  const resetFilters = () => {
    setStatusFilter('all');
    setTypeFilter('all');
    setLocationFilter('all');
  };

  const hasActiveFilters = statusFilter !== 'all' || typeFilter !== 'all' || locationFilter !== 'all';

  return (
    <section className="py-16 md:py-24 bg-brand-midnight-blue min-h-screen">
      <Container>
        {/* Filters Section */}
        <div className="mb-16">
          <div className="flex flex-col md:flex-row gap-4 justify-center items-center mb-8">
            <FilterDropdown 
              label="Location" 
              value={locationFilter} 
              options={locations} 
              onChange={setLocationFilter} 
            />
            <FilterDropdown 
              label="Status" 
              value={statusFilter} 
              options={['Ongoing', 'Completed', 'Upcoming']} 
              onChange={(val) => setStatusFilter(val as ProjectStatus | 'all')} 
            />
            <FilterDropdown 
              label="Type" 
              value={typeFilter} 
              options={['Residential', 'Commercial', 'Mixed-Use']} 
              onChange={(val) => setTypeFilter(val as ProjectType | 'all')} 
            />
            
            {hasActiveFilters && (
              <button
                onClick={resetFilters}
                className="px-4 py-2 text-brand-teal-sky hover:text-white transition-colors text-sm font-medium"
              >
                Reset Filters
              </button>
            )}
          </div>
          
          {hasActiveFilters && (
             <p className="text-center text-brand-concrete text-sm">
               Showing {filteredProjects.length} of {projects.length} projects
             </p>
          )}
        </div>

        {/* Projects Grid */}
        {filteredProjects.length > 0 ? (
          <div className="grid grid-cols-1 md:grid-cols-2 gap-8 md:gap-12">
            {filteredProjects.map((project, index) => (
              <motion.div
                key={project.id}
                initial={{ opacity: 0, y: 50 }}
                whileInView={{ opacity: 1, y: 0 }}
                transition={{ duration: 0.5, delay: index * 0.1 }}
                viewport={{ once: true, margin: "-50px" }}
              >
                <ProjectCard project={project} variant="stack" />
              </motion.div>
            ))}
          </div>
        ) : (
          <div className="text-center py-16">
            <svg className="w-16 h-16 mx-auto text-brand-dutch-blue mb-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={1.5} d="M19 21V5a2 2 0 00-2-2H7a2 2 0 00-2 2v16m14 0h2m-2 0h-5m-9 0H3m2 0h5M9 7h1m-1 4h1m4-4h1m-1 4h1m-5 10v-5a1 1 0 011-1h2a1 1 0 011 1v5m-4 0h4" />
            </svg>
            <h3 className="text-xl font-semibold text-white mb-2">
              No projects found
            </h3>
            <p className="text-brand-concrete mb-6">
              Try adjusting your filters to see more projects
            </p>
            <button
              onClick={resetFilters}
              className="px-6 py-2 bg-brand-dutch-blue text-white font-medium rounded-lg hover:bg-brand-teal-sky transition-colors"
            >
              Reset Filters
            </button>
          </div>
        )}
      </Container>
    </section>
  );
}
