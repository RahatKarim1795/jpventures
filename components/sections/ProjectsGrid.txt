'use client';

import { ProjectWithFeaturedImage } from '@/lib/types';
import ProjectCard from '@/components/ui/ProjectCard';
import SteppedEdge from '@/components/effects/SteppedEdge';
import { useEffect, useState, useRef, forwardRef } from 'react';

interface ProjectsGridProps {
  projects: ProjectWithFeaturedImage[];
  title?: string;
  description?: string;
  limitToOneRow?: boolean;
  showViewAllLink?: boolean;
}

const ProjectsGrid = forwardRef<HTMLElement, ProjectsGridProps>(({ 
  projects, 
  title = "Our Projects",
  description,
  limitToOneRow = false,
  showViewAllLink = false,
}, ref) => {
  const [displayProjects, setDisplayProjects] = useState(projects);
  const [visibleCards, setVisibleCards] = useState<Set<number>>(new Set());
  const cardRefs = useRef<(HTMLDivElement | null)[]>([]);

  useEffect(() => {
    if (limitToOneRow) {
      // Limit based on viewport width
      const handleResize = () => {
        const width = window.innerWidth;
        let limit = 5; // default/max
        
        if (width < 640) {
          limit = 1; // mobile
        } else if (width < 1024) {
          limit = 2; // tablet
        } else if (width < 1280) {
          limit = 3; // small desktop
        } else if (width < 1536) {
          limit = 4; // large desktop
        }
        // >= 1536 is 5
        
        setDisplayProjects(projects.slice(0, limit));
      };
      
      // Initial set
      handleResize();
      
      // Listen for resize
      window.addEventListener('resize', handleResize);
      return () => window.removeEventListener('resize', handleResize);
    } else {
      setDisplayProjects(projects);
    }
  }, [projects, limitToOneRow]);

  // Title animation state
  const [titleVisible, setTitleVisible] = useState(false);
  const titleRef = useRef<HTMLDivElement>(null);

  // Intersection Observer for title animations
  useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            setTitleVisible(true);
            observer.unobserve(entry.target);
          }
        });
      },
      { threshold: 0.2 }
    );

    if (titleRef.current) {
      observer.observe(titleRef.current);
    }

    return () => observer.disconnect();
  }, []);

  // Intersection Observer for card animations
  useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            const index = Number(entry.target.getAttribute('data-card-index'));
            setVisibleCards((prev) => new Set(prev).add(index));
            observer.unobserve(entry.target);
          }
        });
      },
      {
        threshold: 0.2,
      }
    );

    cardRefs.current.forEach((ref) => {
      if (ref) observer.observe(ref);
    });

    return () => observer.disconnect();
  }, [displayProjects]);

  if (projects.length === 0) {
    return null;
  }

  return (
    <section ref={ref} className="relative bg-brand-midnight-blue z-30 overflow-visible pb-8">
      <SteppedEdge />
      <div className="relative z-10 pt-40">
        <div className="w-full px-4 md:px-8">
          
          {/* Section Header */}
          <div 
            ref={titleRef}
            className={`text-left mb-12 transition-all duration-1000 ease-out transform ${titleVisible ? 'opacity-100 translate-x-0' : 'opacity-0 -translate-x-20'}`}
          >
            {/* Badge */}
            <div className="inline-flex items-center space-x-2 bg-white px-4 py-2 rounded-full mb-6">
              <div className="w-2 h-2 bg-brand-dutch-blue rounded-full animate-pulse"></div>
              <span className="text-brand-teal-sky font-semibold text-sm uppercase tracking-wider">
                Our Projects
              </span>
            </div>
            
            <h2 className="text-4xl md:text-5xl lg:text-6xl font-bold text-brand-concrete mb-4">
              {title}
            </h2>
            {description && (
              <p className="text-xl text-brand-concrete max-w-2xl">
                {description}
              </p>
            )}
          </div>

          {/* Projects Grid */}
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-5 gap-6 md:gap-8">
            {displayProjects.map((project, index) => (
              <div
                key={project.id}
                ref={(el) => { cardRefs.current[index] = el; }}
                data-card-index={index}
                className={`card-hidden ${visibleCards.has(index) ? 'animate-slide-up-bounce' : ''}`}
                style={visibleCards.has(index) ? { animationDelay: `${index * 0.15}s` } : undefined}
              >
                <ProjectCard project={project} />
              </div>
            ))}
          </div>

          {/* View All Link */}
          {showViewAllLink && (
            <div className="text-center mt-12">
              <a
                href="/projects"
                className="inline-flex items-center px-8 py-3 bg-brand-dutch-blue text-white font-semibold rounded-lg hover:bg-brand-noir-blue transition-colors duration-300"
              >
                View All Projects
                <svg className="w-5 h-5 ml-2" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M17 8l4 4m0 0l-4 4m4-4H3" />
                </svg>
              </a>
            </div>
          )}
        </div>
      </div>
    </section>
  );
});

ProjectsGrid.displayName = 'ProjectsGrid';

export default ProjectsGrid;
