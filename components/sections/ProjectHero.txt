'use client';

import Image from 'next/image';
import { ProjectWithImages, FileSystemImage, ProjectImage } from '@/lib/types';
import { motion } from 'motion/react';
import { useScroll, useTransform } from 'motion/react';
import { useRef } from 'react';

interface ProjectHeroProps {
  project: ProjectWithImages;
  heroImage?: FileSystemImage | null;
}

// Type guard to check if an image is a ProjectImage (from DB)
function isProjectImage(img: ProjectImage | FileSystemImage): img is ProjectImage {
  return 'filename' in img;
}

export default function ProjectHero({ project, heroImage }: ProjectHeroProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const { scrollYProgress } = useScroll({
    target: containerRef,
    offset: ["start start", "end start"]
  });

  const y = useTransform(scrollYProgress, [0, 1], ["0%", "50%"]);
  const opacity = useTransform(scrollYProgress, [0, 0.8], [1, 0]);

  // Get hero image path
  let imagePath = '/images/placeholder-project.jpg';
  let altText = project.title;

  if (heroImage) {
    // Use file system image if provided
    imagePath = heroImage.src;
    altText = heroImage.alt;
  } else {
    // Fallback to DB images
    // We cast to ensure we are working with the array type we expect from DB results
    const images = project.images as (ProjectImage | FileSystemImage)[];
    
    if (images && images.length > 0) {
      const foundImage = images.find(img => img.isFeatured) || images[0];
      
      if (foundImage) {
        if (isProjectImage(foundImage)) {
          // Handle DB Image
          imagePath = foundImage.filename.startsWith('/') 
            ? foundImage.filename 
            : `/uploads/projects/${project.id}/${foundImage.filename}`;
          altText = foundImage.altText || project.title;
        } else {
          // Handle FileSystemImage (if it somehow ended up in project.images)
          imagePath = foundImage.src;
          altText = foundImage.alt;
        }
      }
    }
  }

  return (
    <section ref={containerRef} className="relative h-screen w-full overflow-hidden">
      {/* Parallax Image Background */}
      <motion.div 
        style={{ y, opacity }}
        className="absolute inset-0 w-full h-full"
      >
        <Image
          src={imagePath}
          alt={altText}
          fill
          className="object-cover"
          priority
          quality={90}
        />
        <div className="absolute inset-0 bg-gradient-to-b from-black/30 via-black/40 to-brand-midnight-blue" />
      </motion.div>

      {/* Content */}
      <div className="relative h-full flex items-center justify-center text-center px-4">
        <div className="max-w-5xl mx-auto">
          <motion.div
            initial={{ opacity: 0, y: 30 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.8, delay: 0.2 }}
          >
            <span className="inline-block px-4 py-1.5 mb-6 rounded-full bg-brand-dutch-blue/80 text-white text-sm font-medium tracking-wider uppercase backdrop-blur-sm border border-white/10">
              {project.status}
            </span>
            <h1 className="text-5xl md:text-7xl lg:text-8xl font-bold text-white mb-6 tracking-tight">
              {project.title}
            </h1>
            <p className="text-xl md:text-2xl text-gray-200 max-w-2xl mx-auto font-light">
              {project.address}
            </p>
          </motion.div>
        </div>
      </div>

      {/* Scroll Indicator */}
      <motion.div 
        initial={{ opacity: 0 }}
        animate={{ opacity: 1 }}
        transition={{ delay: 1, duration: 1 }}
        className="absolute bottom-12 left-1/2 -translate-x-1/2 flex flex-col items-center text-white/50"
      >
        <span className="text-xs uppercase tracking-widest mb-2">Scroll</span>
        <div className="w-px h-12 bg-gradient-to-b from-white/50 to-transparent" />
      </motion.div>
    </section>
  );
}
