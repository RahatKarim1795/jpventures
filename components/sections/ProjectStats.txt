'use client';

import { ProjectWithImages, ProjectImage, FileSystemImage } from '@/lib/types';
import Image from 'next/image';
import Container from '@/components/ui/Container';
import { motion } from 'motion/react';
import { 
  AreaChart, 
  Building2, 
  Car, 
  Compass, 
  Home, 
  Layers, 
  MapPin 
} from 'lucide-react';

interface ProjectStatsProps {
  project: ProjectWithImages;
}

// Type guard
function isProjectImage(img: ProjectImage | FileSystemImage): img is ProjectImage {
  return 'imageType' in img;
}

export default function ProjectStats({ project }: ProjectStatsProps) {
  // Normalize images to array
  const images = project.images as (ProjectImage | FileSystemImage)[];
  
  // Find the main image (featured image or one with filename starting with "main.")
  const mainImage = images.find(img => {
    if (isProjectImage(img)) {
      return img.isFeatured || img.filename.toLowerCase().startsWith('main.');
    }
    // FileSystemImage - check isFeatured flag or src containing 'main.'
    return img.isFeatured || img.src.toLowerCase().includes('/main.');
  }) || images[0]; // Fallback to first image if no main found

  // Resolve path and alt
  let imagePath = '/images/placeholder-project.jpg';
  let altText = `${project.title} - Main View`;

  if (mainImage) {
    if (isProjectImage(mainImage)) {
       imagePath = mainImage.filename.startsWith('/') 
        ? mainImage.filename 
        : `/images/projects/${project.id}/${mainImage.filename}`;
       altText = mainImage.altText || `${project.title} - Main View`;
    } else {
       imagePath = mainImage.src;
       altText = mainImage.alt || `${project.title} - Main View`;
    }
  }

  const stats = [
    { label: 'Land Size', value: project.landSize, icon: AreaChart },
    { label: 'Apartment Size', value: project.apartmentSize, icon: Home },
    { label: 'Total Units', value: project.numberOfUnits, icon: Building2 },
    { label: 'Floors', value: project.numberOfFloors, icon: Layers },
    { label: 'Parking', value: project.numberOfParking, icon: Car },
    { label: 'Orientation', value: project.orientation, icon: Compass },
    { label: 'Front Road', value: project.frontRoad, icon: MapPin },
  ].filter(stat => stat.value); // Only show stats that exist

  return (
    <section className="py-16 md:py-24 bg-brand-midnight-blue">
      <Container>
        {/* Description Section */}
        <div className="max-w-4xl mx-auto text-center mb-20">
          <motion.h2 
            initial={{ opacity: 0, y: 20 }}
            whileInView={{ opacity: 1, y: 0 }}
            viewport={{ once: true }}
            className="text-3xl md:text-4xl font-bold text-white mb-8"
          >
            Project Overview
          </motion.h2>
          <motion.p 
            initial={{ opacity: 0, y: 20 }}
            whileInView={{ opacity: 1, y: 0 }}
            viewport={{ once: true }}
            transition={{ delay: 0.1 }}
            className="text-lg text-brand-concrete leading-relaxed"
          >
            {project.description || "Experience luxury living with thoughtful design and premium amenities."}
          </motion.p>
        </div>

        <div className="grid grid-cols-1 lg:grid-cols-2 gap-12 items-center">
          {/* Big Card (Left Side) */}
          <motion.div 
            initial={{ opacity: 0, x: -50 }}
            whileInView={{ opacity: 1, x: 0 }}
            viewport={{ once: true }}
            transition={{ duration: 0.6 }}
            className="relative h-[600px] w-full rounded-2xl overflow-hidden shadow-2xl group"
          >
            <Image
              src={imagePath}
              alt={altText}
              fill
              className="object-cover transition-transform duration-700 group-hover:scale-105"
            />
            <div className="absolute inset-0 bg-black/20 group-hover:bg-black/0 transition-colors duration-500" />
            
            {/* Floating Label */}
            <div className="absolute bottom-8 left-8 bg-white/90 backdrop-blur-md px-6 py-3 rounded-lg shadow-lg">
              <p className="text-brand-midnight-blue font-semibold text-lg">
                {project.type} Residence
              </p>
            </div>
          </motion.div>

          {/* Stats Grid (Right Side) */}
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-6">
            {stats.map((stat, index) => (
              <motion.div
                key={stat.label}
                initial={{ opacity: 0, y: 20 }}
                whileInView={{ opacity: 1, y: 0 }}
                viewport={{ once: true }}
                transition={{ duration: 0.4, delay: index * 0.1 }}
                className="bg-white/5 p-6 rounded-xl hover:bg-white/10 hover:shadow-lg transition-all duration-300 border border-white/10"
              >
                <div className="flex items-start gap-4">
                  <div className="p-3 bg-brand-teal-sky/20 rounded-lg text-brand-teal-sky">
                    <stat.icon size={24} />
                  </div>
                  <div>
                    <p className="text-brand-concrete text-sm mb-1 font-medium uppercase tracking-wide">
                      {stat.label}
                    </p>
                    <p className="text-white font-bold text-lg">
                      {stat.value}
                    </p>
                  </div>
                </div>
              </motion.div>
            ))}
          </div>
        </div>
      </Container>
    </section>
  );
}
